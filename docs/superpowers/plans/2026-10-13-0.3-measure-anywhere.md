# 0.3 Milestone 1: Measure Anywhere Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let a measurement go anywhere in a circuit, add a Reset gate and one-bit conditions ("run this gate only if q1 measured 1"), simulate them exactly with a density-matrix engine of up to 8 qubits, and export them to Qiskit, Cirq, pyQuil and OpenQASM.

**Architecture:** The circuit model gains `RESET` and an optional `condition` on an operation. A circuit that resets a qubit or uses one after measuring it is "dynamic": `simulate` sends it to a new density-matrix engine (`src/sim/density.ts`), which reuses the state-vector kernels by treating ρ as a 2n-qubit vector. Every other circuit keeps the current state-vector engine. The interface reads both through one `QuantumState` type. When a density matrix is actually pure, `pureVector` recovers its amplitudes, so the view only changes once the state is really a mix. References come from Qiskit's `DensityMatrix`.

**Tech Stack:** TypeScript, React, Zustand, Vitest, Playwright, and Python in the quantum-lab image (Qiskit 2.5.2 with Aer, Cirq 1.7.0, pyQuil 4.22.0).

**Spec:** `PLAN.md`, section "Version 0.3 plan" (In, Technical design, and Milestones item 1). Project rules: `CLAUDE.md`.

/** The qubits an operation acts on, plus the one its condition reads. */
export function referencedQubits(op: Operation): number[] {
  return op.condition ? [...operationQubits(op), op.condition.qubit] : operationQubits(op);
}

/** Qubits measured before `column` and left alone since, which a gate in `column` can be conditioned on. */
export function readableQubits(circuit: Circuit, column: number): number[] {
  const readable = new Set<number>();
  for (const ops of circuit.columns.slice(0, column)) {
    ops.flatMap(operationQubits).forEach((q) => readable.delete(q));
    ops.forEach((op) => op.gate === 'M' && op.targets.forEach((q) => readable.add(q)));
  }
  return [...readable].sort((a, b) => a - b);
}

/**
 * Measurements whose qubit is used again later, as `column:qubit` keys. A terminal measurement
 * isn't listed: it doesn't change the outcome probabilities, so the simulator leaves the state as is.
 */
export function partwayMeasurements(circuit: Circuit): Set<string> {
  const marked = new Set<string>();
  const usedLater = new Set<number>();
  for (let c = circuit.columns.length - 1; c >= 0; c--) {
    for (const op of circuit.columns[c]) {
      if (op.gate === 'M') op.targets.forEach((q) => usedLater.has(q) && marked.add(`${c}:${q}`));
    }
    for (const op of circuit.columns[c]) referencedQubits(op).forEach((q) => usedLater.add(q));
  }
  return marked;
}

/** A dynamic circuit resets a qubit or measures one partway through, so it needs the density-matrix simulator. */
export function isDynamic(circuit: Circuit): boolean {
  return circuit.columns.some((column) => column.some((op) => op.gate === 'RESET')) || partwayMeasurements(circuit).size > 0;
}

/** Deletes the qubit and every gate that touches or reads it; the qubits below move up one row. */
export function removeQubit(circuit: Circuit, qubit: number): EditResult {
  if (circuit.qubits <= 1) return { ok: false, error: 'too-few-qubits' };
  const shift = (q: number) => (q > qubit ? q - 1 : q);
  const columns = circuit.columns.map((column) =>
    column
      .filter((op) => !referencedQubits(op).includes(qubit))
      .map((op) => {
        const moved = remap(op, shift);
        return op.condition ? { ...moved, condition: { ...op.condition, qubit: shift(op.condition.qubit) } } : moved;
      }),
  );
  return finish(circuit.qubits - 1, columns);
}
```

import { partwayMeasurements, type Circuit, type Operation } from '../circuit/model';
import { applySingle, applySwap, conjugate, gateMatrix, mask, type Amplitudes, type Matrix2 } from './kernels';

/** ρ in Qanvas's index order (q0 is the most significant bit), row-major: entry (i, j) is at i·2ⁿ + j. */
export interface DensityMatrix {
  qubits: number;
  re: Float64Array;
  im: Float64Array;
  mixed: true;
}

function zeroDensity(qubits: number): DensityMatrix {
  const size = 1 << (2 * qubits);
  const re = new Float64Array(size);
  re[0] = 1;
  return { qubits, re, im: new Float64Array(size), mixed: true };
}

/**
 * Seen as a 2n-qubit vector, ρ's row index is qubits 0…n−1 and its column index is n…2n−1,
 * so UρU† is U on the row qubits followed by conj(U) on the column qubits.
 */
function asVector(rho: DensityMatrix): Amplitudes {
  return { qubits: 2 * rho.qubits, re: rho.re, im: rho.im };
}

function applyUnitary(rho: DensityMatrix, m: Matrix2, target: number, controls: number[]) {
  const n = rho.qubits;
  applySingle(asVector(rho), m, target, controls);
  applySingle(asVector(rho), conjugate(m), n + target, controls.map((c) => n + c));
}

/** Removes the coherence between the qubit's 0 and 1: the state after measuring it, before anyone looks. */
function dephase(rho: DensityMatrix, qubit: number) {
  const dim = 1 << rho.qubits;
  const m = mask(qubit, rho.qubits);
  for (let i = 0; i < dim; i++) {
    for (let j = 0; j < dim; j++) {
      if ((i ^ j) & m) {
        rho.re[i * dim + j] = 0;
        rho.im[i * dim + j] = 0;
      }
    }
  }
}

/** ρ → K₀ρK₀† + K₁ρK₁† with K₀ = |0⟩⟨0| and K₁ = |0⟩⟨1|: the |1⟩ block moves onto |0⟩. */
function reset(rho: DensityMatrix, qubit: number) {
  const dim = 1 << rho.qubits;
  const m = mask(qubit, rho.qubits);
  for (let i = 0; i < dim; i++) {
    for (let j = 0; j < dim; j++) {
      const k = i * dim + j;
      if ((i | j) & m) {
        rho.re[k] = 0;
        rho.im[k] = 0;
      } else {
        const from = (i | m) * dim + (j | m);
        rho.re[k] += rho.re[from];
        rho.im[k] += rho.im[from];
      }
    }
  }
}

const X = gateMatrix('X');

function applyOperation(rho: DensityMatrix, op: Operation, measuresPartway: (q: number) => boolean) {
  switch (op.gate) {
    case 'M':
      return op.targets.filter(measuresPartway).forEach((q) => dephase(rho, q));
    case 'RESET':
      return reset(rho, op.targets[0]);
    case 'SWAP':
      applySwap(asVector(rho), op.targets[0], op.targets[1]);
      return applySwap(asVector(rho), rho.qubits + op.targets[0], rho.qubits + op.targets[1]);
  }
  const m = gateMatrix(op.gate, op.angle);
  if (!op.condition) return applyUnitary(rho, m, op.targets[0], op.controls ?? []);
  // validateCircuit guarantees the condition qubit was measured (dephased) and left alone since,
  // so controlling on its 0/1 value acts exactly like conditioning on the measured bit.
  const { qubit, value } = op.condition;
  if (value === 0) applyUnitary(rho, X, qubit, []);
  applyUnitary(rho, m, op.targets[0], [qubit]);
  if (value === 0) applyUnitary(rho, X, qubit, []);
}

/** The density matrix before any column, then after each column: `columns.length + 1` matrices. */
export function simulateDensity(circuit: Circuit): DensityMatrix[] {
  const partway = partwayMeasurements(circuit);
  const states = [zeroDensity(circuit.qubits)];
  circuit.columns.forEach((column, c) => {
    const previous = states[states.length - 1];
    const rho: DensityMatrix = { ...previous, re: previous.re.slice(), im: previous.im.slice() };
    for (const op of column) applyOperation(rho, op, (q) => partway.has(`${c}:${q}`));
    states.push(rho);
  });
  return states;
}

/**
 * The qubit's Bloch vector, from its reduced density matrix (a partial trace over the others).
 * With ρ = (I + xX + yY + zZ)/2: z = ρ00 − ρ11 and ρ01 = (x − iy)/2. In a pure state a length
 * below 1 means entanglement; in a mixed one it can also mean a measured, unknown value.
 */
export function blochVector(state: QuantumState, qubit: number): Bloch {
  const { re, im, qubits } = state;
  const dim = 1 << qubits;
  const m = 1 << (qubits - 1 - qubit);
  let p0 = 0;
  let p1 = 0;
  let cr = 0;
  let ci = 0;
  for (let i = 0; i < dim; i++) {
    if (i & m) continue;
    const j = i | m;
    if (isMixed(state)) {
      p0 += re[i * dim + i];
      p1 += re[j * dim + j];
      cr += re[i * dim + j];
      ci += im[i * dim + j];
    } else {
      p0 += re[i] ** 2 + im[i] ** 2;
      p1 += re[j] ** 2 + im[j] ** 2;
      // ρ01 += a_i · conj(a_j)
      cr += re[i] * re[j] + im[i] * im[j];
      ci += im[i] * re[j] - re[i] * im[j];
    }
  }
  return { x: 2 * cr, y: -2 * ci, z: p0 - p1 };
}

/**
 * The state as a vector when it is pure (up to a global phase), or null when it's a mix of outcomes.
 * For pure ρ = ψψ†, column j divided by √ρjj is ψ times the phase of conj(ψj).
 */
export function pureVector(state: QuantumState): StateVector | null {
  if (!isMixed(state)) return state;
  const { qubits, re, im } = state;
  const dim = 1 << qubits;
  let purity = 0;
  for (let k = 0; k < re.length; k++) purity += re[k] ** 2 + im[k] ** 2;
  if (purity < 1 - EPSILON) return null;
  let j = 0;
  for (let i = 1; i < dim; i++) if (re[i * dim + i] > re[j * dim + j]) j = i;
  const scale = 1 / Math.sqrt(re[j * dim + j]);
  const vector = { qubits, re: new Float64Array(dim), im: new Float64Array(dim) };
  for (let i = 0; i < dim; i++) {
    vector.re[i] = re[i * dim + j] * scale;
    vector.im[i] = im[i * dim + j] * scale;
  }
  return vector;
}

/** The state as a row-major density matrix; a pure state ψ becomes ψψ†. */
export function densityOf(state: QuantumState): { re: Float64Array; im: Float64Array } {
  if (isMixed(state)) return state;
  const dim = state.re.length;
  const re = new Float64Array(dim * dim);
  const im = new Float64Array(dim * dim);
  for (let i = 0; i < dim; i++) {
    for (let j = 0; j < dim; j++) {
      re[i * dim + j] = state.re[i] * state.re[j] + state.im[i] * state.im[j];
      im[i * dim + j] = state.im[i] * state.re[j] - state.re[i] * state.im[j];
    }
  }
  return { re, im };
}

export function explainStep(
  circuit: Circuit,
  step: number,
  before: QuantumState,
  after: QuantumState,
  t: Translate,
  language: Language,
): StepExplanation {
  const column = circuit.columns[step - 1];
  const kinds = new Set(column.map(kindOf));
  const title = column.length === 1 ? t(`learn.${kindOf(column[0])}.title` as MessageKey) : t('learn.many.title', { count: column.length });
  const was = pureVector(before);
  const is = pureVector(after);

  const sentences = groups(column).map((group) => describe(group, t, language));
  // A measurement leaves the shown state as it is on purpose, so "nothing changed" would mislead.
  if (was && is && (!kinds.has('M') || kinds.size > 1)) {
    if (sameState(was, is)) sentences.push(t('learn.unchanged'));
    else if (sameUpToPhase(was, is)) sentences.push(t('learn.globalPhase'));
    else sentences.push(...entanglementSentences(was, is, t, language));

  const show = (state: StateVector | null) => (state ? formatState(state) : t('learn.mixed'));
  return { label: columnLabel(column, t, language), title, sentences, before: show(was), after: show(is) };

export function hasConditions(circuit: Circuit): boolean {
  return operations(circuit).some(({ op }) => op.condition !== undefined);

/** Cirq's key for a qubit's nth measurement: each key can be used once, so q1, then q1_2, q1_3… */
export function measurementKey(qubit: number, nth: number): string {
  return nth === 1 ? `q${qubit}` : `q${qubit}_${nth}`;
}

/** How many times each qubit is measured. */
export function measurementCounts(circuit: Circuit): number[] {
  const counts = new Array<number>(circuit.qubits).fill(0);
  for (const { op } of operations(circuit)) if (op.gate === 'M') op.targets.forEach((q) => counts[q]++);
  return counts;
}

  const failures: string[] = [];
  const total = Object.values(counts).reduce((a, b) => a + b, 0);
  if (total !== SHOTS) failures.push(`${label}: ${total} shots instead of ${SHOTS}`);

    measuresAtEnd: measuresAtEnd(circuit),
    shots: SHOTS,
  }));
  const output = runInLab('scripts/run_dynamic.py', input) as { name: string; results: Record<string, { counts: Record<string, number> } | { error: string }>; qiskitQasm: string }[];

  const failures: string[] = [];
  output.forEach(({ name, results, qiskitQasm }, i) => {
    const { circuit } = picked[i];
    const expected = outcomeProbabilities(circuit);
    for (const [run, result] of Object.entries(results)) {
      if ('error' in result) failures.push(`${name} [${run}]: ${result.error}`);
      else failures.push(...countFailures(`${name} [${run}]`, result.counts, expected));
    }
    const imported = importQasm(qiskitQasm);
    if (!imported.ok) {
      failures.push(`${name} [import from Qiskit]: ${JSON.stringify(imported.error)} ${qiskitQasm.slice(0, 200)}`);
      return;
    }
    const a = densityOf(finalState(imported.circuit));
    const b = densityOf(finalState(circuit));
    const worst = Math.max(...Array.from(a.re, (re, k) => Math.hypot(re - b.re[k], a.im[k] - b.im[k])));
    if (!(worst <= TOLERANCE)) failures.push(`${name} [import from Qiskit]: off by ${worst}`);
  });
  expect(failures).toEqual([]);
}, 1_200_000);
```
